% Macro for including pdfs 
% -------------------------------------------------
\newcounter{pdfpage}
\newcommand{\pdfsection}[3]{%
 	\pagestyle{empty}
  	\cleardoublepage
  	\pagestyle{fancy}
  	\phantomsection
  	\addcontentsline{toc}{section}{#1}
  	\markright{#1}
  	\setcounter{pdfpage}{0}%
  	\includepdf[
    	pages=-,
    	offset={\the\marginshift} 0.4mm,
    	pagecommand={%
      		\stepcounter{pdfpage}%
      		\ifnum\value{pdfpage}=1
        		\thispagestyle{sectionfirst}%
      		\else
        		\thispagestyle{fancy}%
      		\fi
    	},
    	scale=1.003,
  	]{#3}%
}
